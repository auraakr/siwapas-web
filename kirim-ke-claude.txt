<?php

namespace App\Domain\Content\Actions;

use App\Domain\Catalog\Models\Facility;
use App\Domain\Content\Enums\DestinationType;
use App\Domain\Content\Enums\PublishStatus;
use App\Domain\Content\Models\Article;
use App\Domain\Content\Models\Destination;
use App\Domain\Content\Models\Faq;
use App\Domain\Content\Models\LocalProduct;
use App\Domain\Content\Services\SiteContact;
use App\Domain\Event\Enums\EventStatus;
use App\Domain\Event\Models\Event;
use Spatie\MediaLibrary\HasMedia;

/**
 * Everything the public landing page shows, as plain arrays for Inertia.
 */
class GetLandingPageContent
{
    public function __construct(private SiteContact $siteContact) {}

    /**
     * @return array<string, mixed>
     */
    public function handle(): array
    {
        return [
            'images' => [
                'hero' => $this->staticImage('hero'),
                'about' => $this->staticImage('about'),
            ],
            'video_url' => config('siwapas.landing.video_url') ?: null,
            'destinations' => $this->destinations(DestinationType::Destination, 4),
            'activities' => $this->destinations(DestinationType::Activity, 3),
            'facilities' => $this->facilities(),
            'products' => $this->products(),
            'events' => $this->events(),
            'articles' => $this->articles(),
            'faqs' => $this->faqs(),
        ];
    }

    /**
     * A photo dropped into public/images/landing (hero.jpg, about.jpg, ...), if present.
     */
    private function staticImage(string $name): ?string
    {
        foreach (['jpg', 'jpeg', 'webp', 'png'] as $extension) {
            if (is_file(public_path("images/landing/{$name}.{$extension}"))) {
                return asset("images/landing/{$name}.{$extension}");
            }
        }

        return null;
    }

    private function imageOf(HasMedia $model): ?string
    {
        return $model->getFirstMediaUrl('default') ?: null;
    }

    /**
     * @return array<int, array<string, mixed>>
     */
    private function destinations(DestinationType $type, int $limit): array
    {
        return Destination::query()
            ->with('media')
            ->where('type', $type)
            ->where('status', PublishStatus::Published)
            ->orderByDesc('is_featured')
            ->orderBy('sort_order')
            ->limit($limit)
            ->get()
            ->map(fn (Destination $destination): array => [
                'id' => $destination->id,
                'name' => $destination->name,
                'summary' => $destination->summary,
                'address' => $destination->address,
                'image' => $this->imageOf($destination),
                'is_featured' => $destination->is_featured,
                'requires_reservation' => $destination->requires_reservation,
            ])
            ->values()
            ->all();
    }

    /**
     * @return array<int, array<string, mixed>>
     */
    private function facilities(): array
    {
        return Facility::query()
            ->where('is_active', true)
            ->orderBy('sort_order')
            ->limit(6)
            ->get()
            ->map(fn (Facility $facility): array => [
                'slug' => $facility->slug,
                'name' => $facility->name,
                'capacity' => $facility->capacity,
                'base_price' => $facility->base_price,
            ])
            ->values()
            ->all();
    }

    /**
     * @return array<int, array<string, mixed>>
     */
    private function products(): array
    {
        return LocalProduct::query()
            ->with('media')
            ->where('is_published', true)
            ->orderBy('sort_order')
            ->limit(6)
            ->get()
            ->map(function (LocalProduct $product): array {
                return [
                    'id' => $product->id,
                    'name' => $product->name,
                    'maker' => $product->maker_name,
                    'description' => $product->description,
                    'image' => $this->imageOf($product),
                    'whatsapp_url' => $this->siteContact->whatsappUrl($product->whatsapp_number),
                ];
            })
            ->values()
            ->all();
    }

    /**
     * @return array<int, array<string, mixed>>
     */
    private function events(): array
    {
        return Event::query()
            ->where('status', EventStatus::Published)
            ->where('starts_at', '>=', now())
            ->orderBy('starts_at')
            ->limit(3)
            ->get()
            ->map(fn (Event $event): array => [
                'slug' => $event->slug,
                'title' => $event->title,
                'starts_at' => $event->starts_at->toIso8601String(),
                'location_name' => $event->location_name,
                'fee' => $event->fee,
            ])
            ->values()
            ->all();
    }

    /**
     * @return array<int, array<string, mixed>>
     */
    private function articles(): array
    {
        return Article::query()
            ->with('media')
            ->where('status', PublishStatus::Published)
            ->where('published_at', '<=', now())
            ->latest('published_at')
            ->limit(3)
            ->get()
            ->map(fn (Article $article): array => [
                'id' => $article->id,
                'title' => $article->title,
                'excerpt' => $article->excerpt,
                'category' => $article->category->value,
                'author' => $article->author_name,
                'published_at' => $article->published_at?->toIso8601String(),
                'image' => $this->imageOf($article),
            ])
            ->values()
            ->all();
    }

    /**
     * @return array<int, array<string, string>>
     */
    private function faqs(): array
    {
        return Faq::query()
            ->where('is_published', true)
            ->orderBy('sort_order')
            ->limit(6)
            ->get()
            ->map(fn (Faq $faq): array => [
                'question' => $faq->question,
                'answer' => $faq->answer,
            ])
            ->values()
            ->all();
    }
}

                    'status' => $event->status->value,
                    'starts_at' => $event->starts_at->toIso8601String(),
                    'fee' => $event->fee,
                    'quota' => $event->quota,
                    'taken' => (int) $event->getAttribute('taken'),
                ]),
        ]);
    }

    public function create(): Response
    {
        return Inertia::render('admin/events/form', [
            'event' => null,
            'registrations' => [],
            'statuses' => array_map(fn (EventStatus $status): string => $status->value, EventStatus::cases()),
        ]);
    }

    public function store(SaveEventRequest $request, SaveEvent $saveEvent): RedirectResponse
    {
        $event = $saveEvent->handle(null, $request->validated());

        Inertia::flash('toast', ['type' => 'success', 'message' => 'Acara ditambahkan.']);

        return to_route('admin.events.edit', $event);
    }

    public function edit(Event $event): Response
    {
        $event->load('registrations.participants', 'registrations.order');

        return Inertia::render('admin/events/form', [
            'event' => [
                'id' => $event->id,
                'slug' => $event->slug,
                'title_id' => $event->getTranslation('title', 'id', false),
                'title_en' => $event->getTranslation('title', 'en', false),
                'description_id' => $event->getTranslation('description', 'id', false),
                'description_en' => $event->getTranslation('description', 'en', false),
                'requirements_id' => $event->getTranslation('requirements', 'id', false),
                'requirements_en' => $event->getTranslation('requirements', 'en', false),
                'organizer' => $event->organizer,
                'location_name' => $event->location_name,
                'address' => $event->address,
                'latitude' => $event->latitude,
                'longitude' => $event->longitude,
                'starts_at' => $this->forInput($event->starts_at),
                'ends_at' => $this->forInput($event->ends_at),
                'fee' => $event->fee,
                'quota' => $event->quota,
                'registration_opens_at' => $this->forInput($event->registration_opens_at),
                'registration_closes_at' => $this->forInput($event->registration_closes_at),
                'status' => $event->status->value,
            ],
            'registrations' => $event->registrations->map(fn (EventRegistration $registration): array => [
                'id' => $registration->id,
                'code' => $registration->registration_code,
                'is_active' => $registration->is_active,
                'participant_count' => $registration->participant_count,
                'participants' => $registration->participants->pluck('name')->all(),
                'order_code' => $registration->order?->code,
                'order_status' => $registration->order?->status->value,
                'contact' => $registration->order ? "{$registration->order->customer_name} ({$registration->order->customer_phone})" : null,
            ])->values(),
            'statuses' => array_map(fn (EventStatus $status): string => $status->value, EventStatus::cases()),
        ]);
    }

    public function update(SaveEventRequest $request, Event $event, SaveEvent $saveEvent): RedirectResponse
    {
        $saveEvent->handle($event, $request->validated());

        Inertia::flash('toast', ['type' => 'success', 'message' => 'Acara diperbarui.']);

        return back();
    }

    public function cancelRegistration(CancelOrderRequest $request, Event $event, EventRegistration $registration, CancelEventRegistration $cancelEventRegistration): RedirectResponse
    {
        abort_unless($registration->event_id === $event->id, 404);

        try {
            $cancelEventRegistration->handle($registration, $request->user(), $request->string('reason')->toString());
        } catch (InvalidOrderTransition) {
            return back()->withErrors(['reason' => 'Pendaftaran ini tidak dapat dibatalkan lagi.']);
        }

        Inertia::flash('toast', ['type' => 'success', 'message' => 'Pendaftaran dibatalkan.']);

        return back();
    }

    private function forInput(?CarbonInterface $value): ?string
    {
        return $value?->copy()->setTimezone(config('siwapas.display_timezone'))->format('Y-m-d\TH:i');
    }

===== \app\Providers\AppServiceProvider.php =====
<?php

namespace App\Providers;

use App\Http\Responses\AdminLogoutResponse;
use Carbon\CarbonImmutable;
use Filament\Http\Responses\Auth\Contracts\LogoutResponse;
use Illuminate\Cache\RateLimiting\Limit;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Date;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\ServiceProvider;
use Illuminate\Validation\Rules\Password;
use Spatie\Translatable\Translatable;

class AppServiceProvider extends ServiceProvider
{
    /**
     * Register any application services.
     */
    public function register(): void
    {
        $this->app->bind(LogoutResponse::class, AdminLogoutResponse::class);
    }

    /**
     * Bootstrap any application services.
     */
    public function boot(): void
    {
        $this->configureDefaults();
        $this->configureRateLimiting();
        $this->configureTranslationFallback();
    }

    /**
     * Indonesian is the primary content language: when a translation is missing for the
     * active locale, show Indonesian, and then any other language that has been written.
     * The admin forms read raw translations, so this never fills the English fields.
     */
    protected function configureTranslationFallback(): void
    {
        $this->app->make(Translatable::class)->fallback(fallbackLocale: 'id', fallbackAny: true);
    }

    /**
     * Configure rate limiters for public write endpoints.
     */
    protected function configureRateLimiting(): void
    {
        RateLimiter::for('public-write', fn (Request $request) => Limit::perMinute(10)->by($request->ip()));
    }

    /**
     * Configure default behaviors for production-ready applications.
     */
    protected function configureDefaults(): void
    {
        Date::use(CarbonImmutable::class);

        DB::prohibitDestructiveCommands(
            app()->isProduction(),
        );

        Password::defaults(fn (): ?Password => app()->isProduction()
            ? Password::min(12)
                ->mixedCase()
                ->letters()
                ->numbers()
                ->symbols()
                ->uncompromised()
            : null,
        );
    }
}

const messages: Record<Locale, Record<MessageKey, string>> = { id, en };

export type Translate = (
    key: MessageKey,
    params?: Record<string, string | number>,
) => string;

export function useTranslation(): { t: Translate; locale: Locale } {
    const { locale } = usePage().props;

    const t: Translate = (key, params) => {
        let text = messages[locale]?.[key] ?? id[key];

        for (const [name, value] of Object.entries(params ?? {})) {
            text = text.replace(`{${name}}`, String(value));
        }

        return text;
    };

    return { t, locale };
}

export const localeNames: Record<Locale, string> = {
    id: 'Bahasa Indonesia',
    en: 'English',
};
